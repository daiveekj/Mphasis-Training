10.List employees working in the 'San Francisco' office.

SELECT  E.firstName, E.lastName, O.city AS officeCity
FROM Employees E

17. List total payments made by each customer.

SELECT T1.customerName, SUM(T2.amount) AS totalPayments
FROM Customers T1
JOIN Payments T2 ON T1.customerNumber = T2.customerNumber
GROUP BY 
T1.customerName
ORDER BY 
totalPayments DESC;

18. Show customers who have made more than 3 payments.

SELECT T1.customerName, COUNT(T2.checkNumber) AS numberOfPayments
FROM Customers T1
JOIN Payments T2 ON T1.customerNumber = T2.customerNumber
GROUP BY T1.customerName
HAVING COUNT(T2.checkNumber) > 3
ORDER BY numberOfPayments DESC;

19. Count the number of employees in each office.

SELECT T2.city, COUNT(T1.employeeNumber) AS numberOfEmployees
FROM Employees T1
JOIN Offices T2 ON T1.officeCode = T2.officeCode
GROUP BY T2.city
ORDER BY numberOfEmployees DESC;

20. List product lines with an average product price greater than 100.

SELECT productLine, AVG(MSRP) AS averageMSRP
FROM Products
GROUP BY 
productLine
HAVING 
AVG(MSRP) > 100
ORDER BY 
averageMSRP DESC;

21. List customers with total order amount greater than 50,000.

SELECT T1.customerName, SUM(T3.quantityOrdered * T3.priceEach) AS totalOrderAmount
FROM Customers T1
JOIN Orders T2 ON T1.customerNumber = T2.customerNumber
JOIN OrderDetails T3 ON T2.orderNumber = T3.orderNumber
GROUP BY 
T1.customerName
HAVING 
totalOrderAmount > 50000
ORDER BY 
totalOrderAmount DESC;

22. Show total quantity ordered for each product code.

SELECT productCode, SUM(quantityOrdered) AS totalQuantityOrdered

23. Display number of orders for each order status.

SELECT status, COUNT(orderNumber) AS numberOfOrders

24. List employees who manage more than 2 other employees.
SELECT T1.firstName, T1.lastName, COUNT(T2.employeeNumber) AS managedEmployees
FROM Employees T1 -- The Manager
JOIN Employees T2 ON T1.employeeNumber = T2.reportsTo 
GROUP BY T1.employeeNumber, T1.firstName, T1.lastName
HAVING COUNT(T2.employeeNumber) > 2
ORDER BY managedEmployees DESC;
