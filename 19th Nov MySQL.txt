33. List all orders along with customer names (INNER JOIN).
SELECT 
    T1.orderNumber, T2.customerName, T1.orderDate
FROM 
    Orders T1
INNER JOIN 
    Customers T2 ON T1.customerNumber = T2.customerNumber;

34. Show employees and their managers using a self join.

SELECT 
    E.firstName AS EmployeeFirstName, 
    E.lastName AS EmployeeLastName, 
    M.firstName AS ManagerFirstName, 
    M.lastName AS ManagerLastName
FROM 
    Employees E -- E is the Employee table alias
INNER JOIN 
    Employees M ON E.reportsTo = M.employeeNumber; 

35. List all products and their order quantities (join products and orderdetails).
SELECT 
    T1.productName, SUM(T2.quantityOrdered) AS totalQuantityOrdered
FROM 
    Products T1
INNER JOIN 
    OrderDetails T2 ON T1.productCode = T2.productCode
GROUP BY 
    T1.productName
ORDER BY 
    totalQuantityOrdered DESC;

36. Show customers who have not placed any orders (LEFT JOIN).

FROM 
    Customers T1
LEFT JOIN

43. Find customers who have made the highest total payment.

SELECT 
    T1.customerName, SUM(T2.amount) AS totalPaid
FROM 
    Customers T1
JOIN 
    Payments T2 ON T1.customerNumber = T2.customerNumber
GROUP BY 
    T1.customerName
HAVING 
    SUM(T2.amount) = (
        -- Subquery: Find the single highest total payment amount
        SELECT MAX(TotalAmount) 
        FROM (
            SELECT SUM(amount) AS TotalAmount 
            FROM Payments 
            GROUP BY customerNumber
        ) AS SubqueryMax
    );

    )

    T1.customerName

        FROM Payments
    );

49. Find the employee(s) with the highest number of customers reporting to them.

SELECT 
    T1.firstName, T1.lastName, COUNT(T2.customerNumber) AS customersManaged
FROM 
    Employees T1 -- The Sales Rep
INNER JOIN 
    Customers T2 ON T1.employeeNumber = T2.salesRepEmployeeNumber
GROUP BY 
    T1.employeeNumber, T1.firstName, T1.lastName
HAVING 
    COUNT(T2.customerNumber) = (
     
        SELECT MAX(CustomerCount)
        FROM (
            SELECT COUNT(customerNumber) AS CustomerCount
            FROM Customers
            GROUP BY salesRepEmployeeNumber
        ) AS SubqueryMax
    );

51. Show customers who ordered the same product more than once.

SELECT DISTINCT
    T1.customerName
FROM 
    Customers T1
INNER JOIN 
    Orders T2 ON T1.customerNumber = T2.customerNumber
INNER JOIN 
    OrderDetails T3 ON T2.orderNumber = T3.orderNumber
GROUP BY 
    T1.customerName, T3.productCode
HAVING 
    COUNT(T3.orderNumber) > 1;
